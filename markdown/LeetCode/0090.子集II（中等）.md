解法1：标记排序数组回溯

时间复杂度 
$$
O(n*2^n)
$$
java

```java
public List<List<Integer>> subsetsWithDup(int[] nums) {
    List<List<Integer>> result = new LinkedList<>();
    Arrays.sort(nums);
    appendItem(0, nums, new boolean[nums.length], new ArrayDeque<>(), result);
    return result;
}

public void appendItem(int level, int[] nums, boolean[] appended, ArrayDeque<Integer> item, List<List<Integer>> result) {
    if (level == nums.length) {
        result.add(new ArrayList<>(item));
        return;
    }
    if (level == 0 || nums[level - 1] != nums[level] || appended[level - 1]) {
        item.push(nums[level]);
        appended[level] = true;
        appendItem(level + 1, nums, appended, item, result);
        appended[level] = false;
        item.pop();
    }
    appendItem(level + 1, nums, appended, item, result);

}
```



#### 解法2：排序+位运算+暴力去重

由于 数字大小为最小为-10，最大为10 范围为21 占5个二进制位，数量最大为10

即数组最大可能为5*10=50 个二进制位，小于64位的长整型，可以将整个子集做位运算，求出唯一的哈希值。

将原始数组排序，从小到大依次选取元素。将每个位置的数组再次加入已有集合，若重复则抛弃，放弃进入集合。

由于给定元素较少，这种暴力解法相较于DFS性能更优

时间复杂度
$$
O(n*2^n)
$$
go

```go
func subsetsWithDup(nums []int) [][]int {
	sort.Ints(nums)
	result := [][]int{{}}
	numSet := make(map[int64]struct{})
	for i := 0; i < len(nums); i++ {
		nr := make([][]int, 0)
		for _, o := range result {
			nr = append(nr, append([]int{nums[i]}, o...))
		}
		for _, n := range nr {
			hash := subsetsWithDupHash(n)
			if _, ok := numSet[hash]; !ok {
				result = append(result, n)
				numSet[hash] = struct{}{}
			}
		}
	}
	return result
}

func subsetsWithDupHash(nums []int) int64 {
	var result int64
	for _, n := range nums {
		result = (result << 5) | int64(n+10)
	}
	return result
}
```
